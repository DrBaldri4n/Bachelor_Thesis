\begin{lstlisting}[caption={Create CSV file and define the correct PromQL query}]
csv_file = 'monitor_lstm_predictions.csv'
header = ['timestamp'] + FEATURES + ['benign_prob', 'attack_prob', 'prediction']

# Function to query a single metric value from Prometheus
def get_metric_value(prom, feat):
    base_feat = feat.replace('-', '_')

    # Try without "perf_" prefix
    try:
        data = prom.get_current_metric_value(metric_name=f"sum({base_feat})")
        df = MetricSnapshotDataFrame(data)
        if not df.empty and 'value' in df.columns:
            return float(df.iloc[0]['value'])
    except Exception:
        pass

    # Try with "perf_" prefix as fallback
    try:
        data = prom.get_current_metric_value(metric_name=f"sum(perf_{base_feat})")
        df = MetricSnapshotDataFrame(data)
        if not df.empty and 'value' in df.columns:
            return float(df.iloc[0]['value'])
    except Exception:
        pass

    return np.nan
\end{lstlisting}

\begin{lstlisting}[caption={Get metric values and log them into the CSV file}]
while True:
    # 1. Query metric values
    values_array = [get_metric_value(prom, feat) for feat in FEATURES]
    
    # 2. Log to CSV
    row = [datetime.now().isoformat()] + values_array + [benign_prob, attack_prob, prediction]
    with open(csv_file, 'a', newline='') as f:
        csv.writer(f).writerow(row)

    time.sleep(5)
\end{lstlisting}
